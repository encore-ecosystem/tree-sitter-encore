#cfg(any(target_os = "linux", target_os = "macos"))
static RENDER_PROFILE: Profile = Profile{1_u32}

#attr(decorator)
fn profile[Args](func: Callable[Args, u32], args: Args) -> u32 {
    ret func(args)
}

@RENDER_PROFILE.profile("draw_frame")
async fn draw_frame() -> u32 {
    let pending = spawn worker()
    ret await pending
}
