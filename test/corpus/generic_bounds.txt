trait Read[T: Debug] { fn read(self: Self) -> T }
impl[T: Debug] Read[T] for Buffer[T] {
    fn read(self: Self) -> T { ret self.value }
}
fn choose[T: Debug](value: T, enabled: bool) -> T {
    let selected = if enabled { value } else { ret value }
    ret selected
}
